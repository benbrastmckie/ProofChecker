/-! ## The proposed abstraction: a truth relation at a point -/

/-- The class abstracting "a truth predicate at a point (model, history, time)".
For a language whose truth recursion carries extra parameters (L⋆'s stored-time vector),
`sat` is the extra-parameter-universally-closed form. -/
class PointTruth (L : Type) where
  sat : ∀ {F : TaskFrame}, TaskModel F → ConvexHistory F → F.Duration → L → Prop

namespace PointTruth

variable {L : Type} [PointTruth L]

def ValidOn (F : TaskFrame) (φ : L) : Prop :=
  ∀ (M : TaskModel F) (τ : TaskFrame.HF F) (x : F.Duration), sat M τ.val x φ

def ValidOnFrames (P : TaskFrame → Prop) (φ : L) : Prop :=
  ∀ F : TaskFrame, P F → ValidOn F φ

def ValidIn (fc : ProofSystem.FrameClass) (φ : L) : Prop :=
  ValidOnFrames fc.Sat φ

def Valid (φ : L) : Prop := ValidIn ProofSystem.FrameClass.Base φ

theorem ValidOn.of_forall_total {F : TaskFrame} {φ : L}
    (h : ∀ (M : TaskModel F) (τ : ConvexHistory F), τ.IsTotal → ∀ x : F.Duration,
           sat M τ x φ) : ValidOn F φ :=
  fun M τ x => h M τ.val τ.property x

theorem ValidOn.apply_total {F : TaskFrame} {φ : L} (h : ValidOn F φ)
    (M : TaskModel F) (τ : ConvexHistory F) (hτ : τ.IsTotal) (x : F.Duration) :
    sat M τ x φ := h M ⟨τ, hτ⟩ x

theorem ValidOnFrames.mono {P Q : TaskFrame → Prop} {φ : L} (h : ∀ F, Q F → P F)
    (hP : ValidOnFrames P φ) : ValidOnFrames Q φ := fun F hF => hP F (h F hF)

theorem ValidIn.mono {fc₁ fc₂ : ProofSystem.FrameClass} {φ : L} (h : fc₁ ≤ fc₂)
    (hv : ValidIn fc₁ φ) : ValidIn fc₂ φ :=
  ValidOnFrames.mono (fun _ => ProofSystem.FrameClass.Sat.anti h) hv

theorem ValidOnFrames.of_forall_total {P : TaskFrame → Prop} {φ : L}
    (h : ∀ (F : TaskFrame), P F → ∀ (M : TaskModel F) (τ : ConvexHistory F),
           τ.IsTotal → ∀ x : F.Duration, sat M τ x φ) : ValidOnFrames P φ :=
  fun F hF M τ x => h F hF M τ.val τ.property x

theorem ValidOnFrames.apply_total {P : TaskFrame → Prop} {φ : L} (h : ValidOnFrames P φ)
    (F : TaskFrame) (hF : P F) (M : TaskModel F) (τ : ConvexHistory F)
    (hτ : τ.IsTotal) (x : F.Duration) : sat M τ x φ :=
  h F hF M ⟨τ, hτ⟩ x

theorem ValidIn.of_forall_total {fc : ProofSystem.FrameClass} {φ : L}
    (h : ∀ (F : TaskFrame), fc.Sat F → ∀ (M : TaskModel F) (τ : ConvexHistory F),
           τ.IsTotal → ∀ x : F.Duration, sat M τ x φ) : ValidIn fc φ :=
  ValidOnFrames.of_forall_total h

theorem ValidIn.apply_total {fc : ProofSystem.FrameClass} {φ : L} (h : ValidIn fc φ)
    (F : TaskFrame) (hF : fc.Sat F) (M : TaskModel F) (τ : ConvexHistory F)
    (hτ : τ.IsTotal) (x : F.Duration) : sat M τ x φ :=
  ValidOnFrames.apply_total h F hF M τ hτ x

theorem Valid.of_forall_total {φ : L}
    (h : ∀ (F : TaskFrame) (M : TaskModel F) (τ : ConvexHistory F), τ.IsTotal →
           ∀ x : F.Duration, sat M τ x φ) : Valid φ :=
  fun F _ M τ x => h F M τ.val τ.property x

theorem Valid.apply {φ : L} (h : Valid φ) (F : TaskFrame) (M : TaskModel F)
    (τ : ConvexHistory F) (hτ : τ.IsTotal) (x : F.Duration) : sat M τ x φ :=
  h F trivial M ⟨τ, hτ⟩ x

theorem ValidOnFrames.of_not {P : TaskFrame → Prop} {φ : L} (h : ¬ ValidOnFrames P φ) :
    ¬ ∀ (F : TaskFrame), P F → ∀ (M : TaskModel F) (τ : ConvexHistory F),
        τ.IsTotal → ∀ x : F.Duration, sat M τ x φ :=
  fun hc => h (ValidOnFrames.of_forall_total hc)

end PointTruth

/-! ## Instances: the four languages, three at (τ,x) and one at (τ,x,v) -/

instance : PointTruth Formula where
  sat M τ t φ := TruthAt M τ t φ
instance : PointTruth MinusFormula where
  sat M τ t φ := MinusTruthAt M τ t φ
instance : PointTruth PlusFormula where
  sat M τ t φ := PlusTruthAt M τ t φ
/-- The L⋆ instance folds the stored-time vector into the point predicate: `v` is the
INNERMOST binder of every L⋆ validity definition, so `∀ v` closes it without changing any
binder telescope downstream. -/
instance : PointTruth StarFormula where
  sat {F} M τ t φ := ∀ v : ℕ → F.Duration, StarTruthAt M τ t v φ
