/-! ## The Boolean core: `⊥`, `→`, `□` at a point with an inert extra environment. -/

class BoolCore (L : Type) where
  Env : TaskFrame → Type
  T : ∀ {F : TaskFrame}, TaskModel F → ConvexHistory F → F.Duration → Env F → L → Prop
  bot : L
  imp : L → L → L
  box : L → L
  bot_clause : ∀ {F : TaskFrame} (M : TaskModel F) (τ : ConvexHistory F) t e, ¬ T M τ t e bot
  imp_clause : ∀ {F : TaskFrame} (M : TaskModel F) (τ : ConvexHistory F) t e (φ ψ : L),
    T M τ t e (imp φ ψ) ↔ (T M τ t e φ → T M τ t e ψ)
  box_clause : ∀ {F : TaskFrame} (M : TaskModel F) (τ : ConvexHistory F) t e (φ : L),
    T M τ t e (box φ) ↔ ∀ σ : ConvexHistory F, σ.IsTotal → T M σ t e φ

namespace BoolCore
variable {L : Type} [BoolCore L] {F : TaskFrame}

def neg (φ : L) : L := imp φ bot
def top : L := imp (bot : L) bot
def and (φ ψ : L) : L := neg (imp φ (neg ψ))
def or (φ ψ : L) : L := imp (neg φ) ψ
def diamond (φ : L) : L := neg (box (neg φ))

theorem neg_iff (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration) (e : Env L F) (φ : L) :
    T M τ t e (neg φ) ↔ ¬ T M τ t e φ := by
  rw [neg, imp_clause]; exact ⟨fun h hφ => bot_clause M τ t e (h hφ),
    fun h hφ => absurd hφ h⟩

theorem top_true (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration) (e : Env L F) :
    T M τ t e (top : L) := by
  rw [top, imp_clause]; exact id

theorem and_iff (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration) (e : Env L F) (φ ψ : L) :
    T M τ t e (and φ ψ) ↔ (T M τ t e φ ∧ T M τ t e ψ) := by
  rw [and, neg_iff, imp_clause, neg_iff]
  constructor
  · intro h
    refine ⟨by_contra fun hφ => h (fun hh => absurd hh hφ), by_contra fun hψ => h (fun _ => hψ)⟩
  · rintro ⟨hφ, hψ⟩ h; exact (h hφ) hψ

theorem or_iff (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration) (e : Env L F) (φ ψ : L) :
    T M τ t e (or φ ψ) ↔ (T M τ t e φ ∨ T M τ t e ψ) := by
  rw [or, imp_clause, neg_iff]
  exact ⟨fun h => by_cases (fun hφ => Or.inl hφ) (fun hφ => Or.inr (h hφ)),
    fun h hn => h.elim (fun hφ => absurd hφ hn) id⟩

theorem diamond_iff (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration) (e : Env L F)
    (φ : L) : T M τ t e (diamond φ) ↔ ∃ σ : ConvexHistory F, σ.IsTotal ∧ T M σ t e φ := by
  rw [diamond, neg_iff, box_clause]
  constructor
  · intro h; by_contra hc; push_neg at hc
    exact h (fun σ hσ => (neg_iff M σ t e φ).mpr (hc σ hσ))
  · rintro ⟨σ, hσ, hφ⟩ h; exact ((neg_iff M σ t e φ).mp (h σ hσ)) hφ

end BoolCore

/-! ## The temporal core: `untl`/`snce` primitive (L, L⁺, L⋆ only -- L⁻ has
`allPast`/`allFuture` primitive instead, which is the FIRST honest boundary). -/

class UntlCore (L : Type) extends BoolCore L where
  untl : L → L → L
  snce : L → L → L
  untl_clause : ∀ {F : TaskFrame} (M : TaskModel F) (τ : ConvexHistory F) t e (ψ φ : L),
    T M τ t e (untl ψ φ) ↔ ∃ s, t < s ∧ T M τ s e φ ∧ ∀ r, t < r → r < s → T M τ r e ψ
  snce_clause : ∀ {F : TaskFrame} (M : TaskModel F) (τ : ConvexHistory F) t e (ψ φ : L),
    T M τ t e (snce ψ φ) ↔ ∃ s, s < t ∧ T M τ s e φ ∧ ∀ r, s < r → r < t → T M τ r e ψ

namespace UntlCore
variable {L : Type} [UntlCore L] {F : TaskFrame}

def someFuture (φ : L) : L := untl BoolCore.top φ
def somePast (φ : L) : L := snce BoolCore.top φ
def allFuture (φ : L) : L := BoolCore.neg (someFuture (BoolCore.neg φ))
def allPast (φ : L) : L := BoolCore.neg (somePast (BoolCore.neg φ))

theorem someFuture_iff (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration)
    (e : BoolCore.Env L F) (φ : L) :
    BoolCore.T M τ t e (someFuture φ) ↔ ∃ s, t < s ∧ BoolCore.T M τ s e φ := by
  rw [someFuture, untl_clause]
  exact ⟨fun ⟨s, hs, hφ, _⟩ => ⟨s, hs, hφ⟩,
    fun ⟨s, hs, hφ⟩ => ⟨s, hs, hφ, fun r _ _ => BoolCore.top_true M τ r e⟩⟩

theorem somePast_iff (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration)
    (e : BoolCore.Env L F) (φ : L) :
    BoolCore.T M τ t e (somePast φ) ↔ ∃ s, s < t ∧ BoolCore.T M τ s e φ := by
  rw [somePast, snce_clause]
  exact ⟨fun ⟨s, hs, hφ, _⟩ => ⟨s, hs, hφ⟩,
    fun ⟨s, hs, hφ⟩ => ⟨s, hs, hφ, fun r _ _ => BoolCore.top_true M τ r e⟩⟩

theorem allFuture_iff (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration)
    (e : BoolCore.Env L F) (φ : L) :
    BoolCore.T M τ t e (allFuture φ) ↔ ∀ s, t < s → BoolCore.T M τ s e φ := by
  rw [allFuture, BoolCore.neg_iff, someFuture_iff]
  constructor
  · intro h s hs; by_contra hc
    exact h ⟨s, hs, (BoolCore.neg_iff M τ s e φ).mpr hc⟩
  · rintro h ⟨s, hs, hn⟩; exact ((BoolCore.neg_iff M τ s e φ).mp hn) (h s hs)

theorem allPast_iff (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration)
    (e : BoolCore.Env L F) (φ : L) :
    BoolCore.T M τ t e (allPast φ) ↔ ∀ s, s < t → BoolCore.T M τ s e φ := by
  rw [allPast, BoolCore.neg_iff, somePast_iff]
  constructor
  · intro h s hs; by_contra hc
    exact h ⟨s, hs, (BoolCore.neg_iff M τ s e φ).mpr hc⟩
  · rintro h ⟨s, hs, hn⟩; exact ((BoolCore.neg_iff M τ s e φ).mp hn) (h s hs)

end UntlCore

/-! ## The `stab` layer: L⁺ and L⋆ only. -/

class StabCore (L : Type) extends UntlCore L where
  stab : L → L
  stab_clause : ∀ {F : TaskFrame} (M : TaskModel F) (τ : ConvexHistory F) t e (φ : L),
    T M τ t e (stab φ) ↔
      ∀ σ : ConvexHistory F, σ.IsTotal → SameStateAt τ σ t → T M σ t e φ

namespace StabCore
variable {L : Type} [StabCore L] {F : TaskFrame}

def dstab (φ : L) : L := BoolCore.neg (stab (BoolCore.neg φ))

theorem dstab_iff (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration)
    (e : BoolCore.Env L F) (φ : L) :
    BoolCore.T M τ t e (dstab φ) ↔
      ∃ σ : ConvexHistory F, σ.IsTotal ∧ SameStateAt τ σ t ∧ BoolCore.T M σ t e φ := by
  rw [dstab, BoolCore.neg_iff, stab_clause]
  constructor
  · intro h; by_contra hc; push_neg at hc
    exact h (fun σ hσ hs => (BoolCore.neg_iff M σ t e φ).mpr (hc σ hσ hs))
  · rintro ⟨σ, hσ, hs, hφ⟩ h; exact ((BoolCore.neg_iff M σ t e φ).mp (h σ hσ hs)) hφ

end StabCore

/-! ## `always`, in both of the two shapes that exist. -/

namespace UntlCore
variable {L : Type} [UntlCore L] {F : TaskFrame}

def always (φ : L) : L := BoolCore.and (allPast φ) (BoolCore.and φ (allFuture φ))

theorem always_iff_tri (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration)
    (e : BoolCore.Env L F) (φ : L) :
    BoolCore.T M τ t e (always φ) ↔
      (∀ s, s < t → BoolCore.T M τ s e φ) ∧ BoolCore.T M τ t e φ ∧
        (∀ s, t < s → BoolCore.T M τ s e φ) := by
  rw [always, BoolCore.and_iff, BoolCore.and_iff, allPast_iff, allFuture_iff]

theorem always_iff_all (M : TaskModel F) (τ : ConvexHistory F) (t : F.Duration)
    (e : BoolCore.Env L F) (φ : L) :
    BoolCore.T M τ t e (always φ) ↔ ∀ s, BoolCore.T M τ s e φ := by
  rw [always_iff_tri]
  constructor
  · rintro ⟨hp, hn, hf⟩ s
    rcases lt_trichotomy s t with h | h | h
    · exact hp s h
    · exact h ▸ hn
    · exact hf s h
  · intro h; exact ⟨fun s _ => h s, h t, fun s _ => h s⟩

end UntlCore

example (φ : Formula) : Formula.always φ = UntlCore.always φ := rfl
example (φ : StarFormula) : StarFormula.always φ = UntlCore.always φ := rfl

    StarTruthAt M τ t v (StarFormula.always φ) ↔
      (∀ s, s < t → StarTruthAt M τ s v φ) ∧ StarTruthAt M τ t v φ ∧
        (∀ s, t < s → StarTruthAt M τ s v φ) :=
